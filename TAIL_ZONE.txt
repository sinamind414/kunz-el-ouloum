203: function inferGlobalUnitIdFromBreadcrumb(breadcrumb: string): number {
204:   const domain1 = breadcrumb.includes('المجال الأول');
205:   const domain2 = breadcrumb.includes('المجال الثاني');
206:   const domain3 = breadcrumb.includes('المجال الثالث');
207: 
208:   const unitNumberMatch = breadcrumb.match(/الوحدة\s+(\d+)/);
209:   const unitNumber = unitNumberMatch
210:     ? Number(unitNumberMatch[1])
211:     : breadcrumb.includes('الوحدة الأولى')
212:       ? 1
213:       : breadcrumb.includes('الوحدة الثانية')
214:         ? 2
215:         : breadcrumb.includes('الوحدة الثالثة')
216:           ? 3
217:           : 1;
218: 
219:   if (domain1) return unitNumber;
220:   if (domain2) return 5 + unitNumber;
221:   if (domain3) return 8 + unitNumber;
222:   return unitNumber;
223: }

225: // Split each bundled EXPERIMENTAL_LESSONS entry into individual lessons.
226: // A new lesson begins whenever a phase contains a `problem` block.
227: // The first sub-lesson keeps the original key (so existing launchers keep working);
228: // subsequent sub-lessons get `${key}_2`, `${key}_3`, ... suffixes.
229: function splitIntoLessons(key: string, lesson: ExperimentalLesson): { key: string; lesson: ExperimentalLesson }[] {
230:   const groups: LessonPhase[][] = [];
231:   let current: LessonPhase[] = [];
232:   for (const phase of lesson.phases) {
233:     const startsNew = phase.blocks.some((b) => b.type === 'problem');
234:     if (startsNew && current.length > 0) {
235:       groups.push(current);
236:       current = [];
237:     }
238:     current.push(phase);
239:   }
240:   if (current.length > 0) groups.push(current);
241: 
242:   return groups.map((phases, i) => {
243:     const firstProblem = phases.flatMap((p) => p.blocks).find((b) => b.type === 'problem');
244:     const useOriginalTitle = groups.length === 1 || i === 0;
245:     const derivedTitle = useOriginalTitle
246:       ? lesson.titleAr
247:       : firstProblem?.title
248:         ? firstProblem.title.replace(/^⚠️\s*/, '').trim()
249:         : lesson.titleAr;
250:     return {
251:       key: groups.length === 1 ? key : i === 0 ? key : `${key}_${i + 1}`,
252:       lesson: {
253:         ...lesson,
254:         titleAr: derivedTitle,
255:         phases,
256:       },
257:     };
258:   });
259: }
260: 
261: const SPLIT_LESSON_ENTRIES = Object.entries(EXPERIMENTAL_LESSONS).flatMap(([key, lesson]) =>
262:   splitIntoLessons(key, lesson)
263: );
264: 
265: export const LESSON_LIBRARY: LessonLibraryItem[] = SPLIT_LESSON_ENTRIES.map(({ key, lesson }) => ({
266:   key,
267:   unitId: inferGlobalUnitIdFromBreadcrumb(lesson.breadcrumb),
268:   ...lesson
269: }));
